const formatDueDate = (due) => {
  if (!due) return "";

  const normalizedDate = String(due).slice(0, 10);

  return /^\d{4}-\d{2}-\d{2}$/.test(normalizedDate)
    ? normalizedDate
    : "";
};

function getDeadlineInfo(
                          due,
                          done,
                          endOfDay = "17:00",
                          dueSoonHours = 48
                        ) {
  if (!due || done) return null;

  // On ne conserve que YYYY-MM-DD, comme formatDueDate()
  const normalizedDate = String(due).slice(0, 10);

  const match = normalizedDate.match(
    /^(\d{4})-(\d{2})-(\d{2})$/
  );

  if (!match) {
    return {
      type: "normal",
      label: null,
    };
  }

  const [, year, month, day] = match;

  // Une tâche est considérée comme due à la fin du jour choisi.
  // Construction en heure LOCALE pour éviter les décalages UTC.
  const [endHour, endMinute] = String(endOfDay)
  .split(":")
  .map(Number);

  const safeHour = Number.isInteger(endHour)
    ? endHour
    : 17;

  const safeMinute = Number.isInteger(endMinute)
    ? endMinute
    : 0;

  const deadline = new Date(
    Number(year),
    Number(month) - 1,
    Number(day),
    safeHour,
    safeMinute,
    0,
    0
  );

  const now = new Date();
  const diffMs = deadline.getTime() - now.getTime();

  const HOUR_MS = 60 * 60 * 1000;
  const DAY_MS = 24 * HOUR_MS;

  // Échéance dépassée
  if (diffMs < 0) {
    const overdueMs = Math.abs(diffMs);
    const days = Math.floor(overdueMs / DAY_MS);
    const hours = Math.floor(
      (overdueMs % DAY_MS) / HOUR_MS
    );

    return {
      type: "overdue",
      label:
        days > 0
          ? `En retard de ${days} j ${hours} h`
          : `En retard de ${hours} h`,
    };
  }

  // Échéance dans moins de 48 h
  const alertHours = Number(dueSoonHours);

  const safeAlertHours =
    Number.isFinite(alertHours) && alertHours > 0
      ? alertHours
      : 48;

  if (diffMs < safeAlertHours * HOUR_MS) {    const days = Math.floor(diffMs / DAY_MS);
    const hours = Math.floor(
      (diffMs % DAY_MS) / HOUR_MS
    );

    return {
      type: "soon",
      label:
        days > 0
          ? `Dans ${days} j ${hours} h`
          : `Dans ${hours} h`,
    };
  }

  return {
    type: "normal",
    label: null,
  };
}

  const [authLoading, setAuthLoading] = useState(true);

  const [email, setEmail] = useState("");
  const [password, setPassword] = useState("");
  const [authError, setAuthError] = useState("");

  const [tasks, setTasks] = useState([]);

  const [settings, setSettings] = useState(DEFAULT_SETTINGS);
  const [settingsLoading, setSettingsLoading] = useState(false);
  const [settingsOpen, setSettingsOpen] = useState(false);
  const [settingsSaving, setSettingsSaving] = useState(false);

  async function loadSettings() {
    if (!session?.user?.id) return null;

    setSettingsLoading(true);
    setError("");

    const { data, error } = await supabase
      .from("user_settings")
      .select(
        "end_of_day, due_soon_hours, trash_retention_days, completed_retention_days"
      )
      .eq("user_id", session.user.id)
      .maybeSingle();

    if (error) {
      setError(
        `Impossible de charger les paramètres : ${error.message}`
      );
      setSettingsLoading(false);
      return null;
    }

    // Première connexion : création des paramètres par défaut
    if (!data) {
      const initialSettings = {
        user_id: session.user.id,
        ...DEFAULT_SETTINGS,
      };

      const { data: created, error: insertError } =
        await supabase
          .from("user_settings")
          .insert(initialSettings)
          .select(
            "end_of_day, due_soon_hours, trash_retention_days, completed_retention_days"
          )
          .single();

      if (insertError) {
        setError(
          `Impossible de créer les paramètres : ${insertError.message}`
        );
        setSettingsLoading(false);
        return null;
      }

      const normalizedSettings = {
        end_of_day: String(
          created.end_of_day
        ).slice(0, 5),
        due_soon_hours: Number(
          created.due_soon_hours
        ),
        trash_retention_days: Number(
          created.trash_retention_days
        ),
        completed_retention_days: Number(
          created.completed_retention_days
        ),
      };

      setSettings(normalizedSettings);
      setSettingsLoading(false);

      return normalizedSettings;
    }

    const normalizedSettings = {
      end_of_day: String(
        data.end_of_day
      ).slice(0, 5),
      due_soon_hours: Number(
        data.due_soon_hours
      ),
      trash_retention_days: Number(
        data.trash_retention_days
      ),
      completed_retention_days: Number(
        data.completed_retention_days
      ),
    };

setSettings(normalizedSettings);
setSettingsLoading(false);

return normalizedSettings;

    return data;
}

    supabase.auth.getSession().then(({ data }) => {
      setSession(data.session);
      setAuthLoading(false);
    });

    const {
      data: { subscription },

      setSession(nextSession);

    return () => subscription.unsubscribe();
  }, []);

  useEffect(() => {
    if (!session?.user) {
      setTasks([]);
      setSettings(DEFAULT_SETTINGS);
      return;

    initializeUserData();

  async function signIn(event) {
    event.preventDefault();

    setAuthError("");
    setAuthLoading(true);

    const { error } = await supabase.auth.signInWithPassword({
      email: email.trim(),
      password,
    });

    if (error) {
      setAuthError(error.message);
    }

    setAuthLoading(false);
  }

  async function signOut() {
    await supabase.auth.signOut();
  }

  async function loadTasks() {
    setLoading(true);
    setError("");

    const { data, error } = await supabase
      .from("tasks")
      .select("*")
      .eq("user_id", session.user.id)
      .order("created_at", { ascending: false });

    if (error) {
      setError(error.message);
    } else {
      setTasks(data ?? []);
    }

    setLoading(false);
  }

  async function purgeExpiredTrash(retentionDays) {
    if (!session?.user?.id) return;

    const days = Number(retentionDays);

    if (!Number.isFinite(days) || days <= 0) {
      return;
    }

    const cutoff = new Date();

    cutoff.setDate(
      cutoff.getDate() - days
    );

    const { error } = await supabase
      .from("tasks")
      .delete()
      .eq("user_id", session.user.id)
      .not("deleted_at", "is", null)
      .lt("deleted_at", cutoff.toISOString());

    if (error) {
      console.error(
        "Erreur pendant la purge de la corbeille :",
        error.message
      );

      setError(
        `La purge automatique n’a pas pu être effectuée : ${error.message}`
      );
    }
  }

  async function purgeExpiredCompleted(retentionDays) {
    if (!session?.user?.id) return;

    const days = Number(retentionDays);

    if (!Number.isFinite(days) || days <= 0) {
      return;
    }

    const cutoff = new Date();

    cutoff.setDate(
      cutoff.getDate() - days
    );

    const { error: purgeError } = await supabase
      .from("tasks")
      .delete()
      .eq("user_id", session.user.id)
      .eq("done", true)
      .is("deleted_at", null)
      .not("completed_at", "is", null)
      .lt("completed_at", cutoff.toISOString());

    if (purgeError) {
      console.error(
        "Erreur pendant la purge des tâches terminées :",
        purgeError.message
      );

      setError(
        `La purge des tâches terminées n’a pas pu être effectuée : ${purgeError.message}`
      );

      return false;
    }

    return true;
  }

  async function initializeUserData() {
    const loadedSettings = await loadSettings();

    if (!loadedSettings) {
      await loadTasks();
      return;
    }

    await purgeExpiredTrash(
      loadedSettings.trash_retention_days
    );

    await purgeExpiredCompleted(
      loadedSettings.completed_retention_days
    );

    await loadTasks();
  }

  async function saveSettings() {
    if (!session?.user?.id) return;

    setSettingsSaving(true);
    setError("");

    const values = {
      user_id: session.user.id,
      end_of_day: String(
        settings.end_of_day
      ).slice(0, 5),
      due_soon_hours: Number(
        settings.due_soon_hours
      ),
      trash_retention_days: Number(
        settings.trash_retention_days
      ),
      completed_retention_days: Number(
        settings.completed_retention_days
      ),
      updated_at: new Date().toISOString(),
    };

    const { data, error } = await supabase
      .from("user_settings")
      .upsert(values, {
        onConflict: "user_id",
      })
      .select(
        "end_of_day, due_soon_hours, trash_retention_days, completed_retention_days"
      )
      .single();

    if (error) {
      setError(
        `Impossible d’enregistrer les paramètres : ${error.message}`
      );
      setSettingsSaving(false);
      return;
    }

    setSettings(data);
    setSettingsOpen(false);
    setSettingsSaving(false);

    // Appliquer immédiatement le nouveau délai de rétention
    await purgeExpiredTrash(
      data.trash_retention_days
    );

    await purgeExpiredCompleted(
      data.completed_retention_days
    );

    await loadTasks();
  }

    setEditingId(null);
    setForm({
      ...EMPTY_FORM,
      quadrant,
    });
    setModalOpen(true);
  }

  function openEditTask(task) {
    setEditingId(task.id);

    setForm({
      title: task.title,
      quadrant: task.quadrant,
      due: formatDueDate(task.due) ?? "",
      tag: task.tag ?? "Perso",
      notes: task.notes ?? "",
    });

    setModalOpen(true);
  }

  async function saveTask() {
    if (!form.title.trim()) return;

    setError("");

    const values = {
      title: form.title.trim(),
      quadrant: form.quadrant,
      due: form.due || null,
      tag: form.tag,
      notes: form.notes.trim() || null,
      updated_at: new Date().toISOString(),
    };

    let result;

    if (editingId) {
      result = await supabase
        .from("tasks")
        .update(values)
        .eq("id", editingId);
    } else {
      result = await supabase.from("tasks").insert({
        ...values,
        user_id: session.user.id,
        done: false,
      });
    }

    if (result.error) {
      setError(result.error.message);
      return;
    }

    setModalOpen(false);
    setForm(EMPTY_FORM);

    await loadTasks();
  }

  async function updateTask(id, changes) {
    setError("");

    const { error } = await supabase
      .from("tasks")
      .update({
        ...changes,
        updated_at: new Date().toISOString(),
      })
      .eq("id", id);

    if (error) {
      setError(error.message);
      return;
    }

    await loadTasks();
  }

  async function moveToTrash(task) {
    await updateTask(task.id, {
      deleted_at: new Date().toISOString(),
    });
  }

  async function restoreTask(task) {
    await updateTask(task.id, {
      deleted_at: null,
    });
  }

  async function deletePermanently(task) {
    const { error } = await supabase
      .from("tasks")
      .delete()
      .eq("id", task.id);

    if (error) {
      setError(error.message);
      return;
    }

    await loadTasks();
  }

    event.preventDefault();

    const id = event.dataTransfer.getData("taskId");

    if (!id) return;

    await updateTask(id, { quadrant });
  }

    const q = search.trim().toLowerCase();

    return tasks.filter((task) => {
      const searchable =
        `${task.title} ${task.notes ?? ""} ${task.tag ?? ""}`.toLowerCase();

      const matchesSearch = !q || searchable.includes(q);

      const matchesTag =
        tagFilter === "Tous" || task.tag === tagFilter;

      return matchesSearch && matchesTag;

  }, [tasks, search, tagFilter]);

  const activeTasks = filteredTasks.filter(
    (task) => !task.done && !task.deleted_at

  const completedTasks = filteredTasks.filter(
    (task) => task.done && !task.deleted_at
  );

  const trashTasks = filteredTasks.filter(
    (task) => Boolean(task.deleted_at)

  const activeCount = tasks.filter(
    (task) => !task.done && !task.deleted_at
  ).length;

  const completedCount = tasks.filter(
    (task) => task.done && !task.deleted_at

  const trashCount = tasks.filter(
    (task) => Boolean(task.deleted_at)
  ).length;

  if (authLoading && !session) {

                      onComplete={() =>

function TaskCard({
  settings,
  task,
  onEdit,
  onComplete,
  onDelete,
}) {

  const deadlineInfo = getDeadlineInfo(
    task.due,
    task.done,
    settings?.end_of_day,
    settings?.due_soon_hours
  );

  function formatDate(date) {
    if (!date) return "Sans échéance";

    const normalizedDate = String(date).slice(0, 10);
    const match = normalizedDate.match(
      /^(\d{4})-(\d{2})-(\d{2})$/
    );

    if (!match) {
      console.warn("Date d'échéance invalide :", date);
      return "Date invalide";
    }

    const [, year, month, day] = match;

    const parsedDate = new Date(
      Number(year),
      Number(month) - 1,
      Number(day),
      12,
      0,
      0
    );

    return new Intl.DateTimeFormat("fr-FR", {
      day: "numeric",
      month: "short",
      year: "numeric",
    }).format(parsedDate);
  }

  return (
    <article
      className="task"
      draggable={!task.done}
      onDragStart={(e) =>
        e.dataTransfer.setData("taskId", task.id)
      }
    >
      <>
      <ReloadPrompt />
      </>
      <button
        className={
          task.done
            ? "task-check task-check-done"
            : "task-check"
        }
        onClick={onComplete}
      >
        {task.done && <Check size={13} />}
      </button>

      <div className="task-content">
        <div
          className={
            task.done
              ? "task-title task-title-done"
              : "task-title"
          }
        >
          {task.title}
        </div>

        {task.notes && (
          <p className="task-notes">
            {task.notes}
          </p>
        )}

        <div className="task-meta">
          <span
            className={
              deadlineInfo?.type === "overdue"
                ? "deadline overdue"
                : deadlineInfo?.type === "soon"
                ? "deadline soon"
                : "deadline"
            }
          >
            <Clock3 size={13} />

            {deadlineInfo?.label || formatDate(task.due)}
          </span>

          <span className="task-tag">
            <Tag size={12} />
            {task.tag}
          </span>
        </div>
      </div>

      <div className="task-actions">
        <button
          className="icon-button"
          onClick={onEdit}
        >
          <Pencil size={15} />
        </button>

        <button
          className="icon-button icon-danger"
          onClick={onDelete}
        >
          <Trash2 size={15} />
        </button>
      </div>
    </article>
  );
}

import { useRegisterSW } from 'virtual:pwa-register/react'

export default function ReloadPrompt() {
  const {
    needRefresh: [needRefresh, setNeedRefresh],
    updateServiceWorker,
  } = useRegisterSW()

  if (!needRefresh) return null

  return (
    <div
      style={{
        position: 'fixed',
        bottom: '20px',
        left: '50%',
        transform: 'translateX(-50%)',
        zIndex: 9999,
        background: '#ffffff',
        color: '#111827',
        padding: '12px 16px',
        borderRadius: '12px',
        boxShadow: '0 4px 20px rgba(0,0,0,0.18)',
        display: 'flex',
        alignItems: 'center',
        gap: '16px',
        border: '1px solid #e5e7eb',
      }}
    >
      <span>Nouvelle version disponible</span>

      <button
        onClick={() => updateServiceWorker(true)}
        style={{
          background: '#047857',
          color: '#ffffff',
          border: 'none',
          borderRadius: '8px',
          padding: '8px 12px',
          fontWeight: '600',
          cursor: 'pointer',
        }}
      >
        Actualiser
      </button>

      <button
        onClick={() => setNeedRefresh(false)}
        aria-label="Fermer"
        style={{
          background: 'transparent',
          border: 'none',
          cursor: 'pointer',
          fontSize: '18px',
          color: '#6b7280',
        }}
      >
        ×
      </button>
    </div>
  )
}
